# OKR-Driven Morning Cockpit Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A morning cockpit (`cockpit.html`) whose hero is the live OKR scoreboard, with a "start here today" triage, project lanes grouped under the KR they serve, and an ad-hoc quick-start lane.

**Architecture:** `okr.py` (in the SecondBrain vault) exposes its data as `okrs.json`; KeystoneOS reads that artifact (no Python at render time), links projects to KRs via an `okr` tag in `projects.json`, and renders a static `cockpit.html` regenerated by the morning cadence. Reuses the existing dashboard file-writing pattern. See spec `docs/superpowers/specs/2026-06-12-okr-cockpit-design.md`.

**Tech Stack:** Python 3 + PyYAML (`okr.py`), PowerShell 7 (`scripts/keystone.ps1`), JSON registry, plain-ps1 `Assert-True` tests.

```python
def json_payload(data: dict) -> dict:
    meta = data.get("meta", {})
    by_obj: dict[str, list] = {}
    for kr in data.get("key_results", []):
        by_obj.setdefault(kr["objective"], []).append(kr)
    objectives = []
    for obj in data.get("objectives", []):
        krs = []
        for kr in by_obj.get(obj["id"], []):
            c = latest(kr) or {}
            krs.append({
                "id": kr["id"],
                "text": kr.get("text", ""),
                "target": kr.get("target", ""),
                "owner": kr.get("owner", ""),
                "coowner": kr.get("coowner"),
                "status": c.get("status"),
                "confidence": c.get("confidence"),
                "next_action": c.get("next_action"),
                "blocker": c.get("blocker"),
                "last_checkin": c.get("date"),
                "trend": [x.get("confidence") for x in (kr.get("checkins") or [])],
            })
        objectives.append({"id": obj["id"], "name": obj["name"], "key_results": krs})
    return {
        "meta": {
            "cycle": meta.get("cycle", ""),
            "gate": str(meta.get("gate", "")),
            "generated": dt.date.today().isoformat(),
        },
        "objectives": objectives,
    }


def write_json(data: dict) -> None:
    OUT_JSON.write_text(
        json.dumps(json_payload(data), indent=2, ensure_ascii=False) + "\n",
        encoding="utf-8",
    )
```

```powershell
function Invoke-OkrLink($ProjectKey) {
    if (-not $ProjectKey) { Write-Host "Usage: .\scripts\keystone.ps1 okr-link <ProjectKey> <KR-id> [<KR-id> ...]"; return }
    $krIds = @($Rest) | Where-Object { $_ }
    if ($krIds.Count -eq 0) { Write-Host "Provide at least one KR id, e.g. okr-link FaceComps 1.2"; return }
    if (-not (Test-Path $ProjectDataPath)) { Write-Host "No projects.json."; return }

    $json = Get-Content -Raw -Path $ProjectDataPath | ConvertFrom-Json
    $projects = @($json.projects)
    $target = $projects | Where-Object { $_.key -ieq $ProjectKey } | Select-Object -First 1
    if (-not $target) { Write-Host "No project found for key '$ProjectKey'."; return }

    $existing = @($target.okr)
    $merged = @($existing + $krIds | Where-Object { $_ } | Select-Object -Unique)
    if ($target.PSObject.Properties.Name -contains 'okr') { $target.okr = $merged }
    else { $target | Add-Member -NotePropertyName okr -NotePropertyValue $merged -Force }

    if ($DryRun) { Write-Host "Dry run: would set $ProjectKey okr = $($merged -join ', ')"; return }
    [pscustomobject]@{ projects = @($projects) } | ConvertTo-Json -Depth 8 | Set-Content -Path $ProjectDataPath -Encoding UTF8
    Write-Host "Linked $ProjectKey -> KR $($merged -join ', ')"

    if (Test-Path (Join-Path $Root '.git')) {
        try {
            Push-Location $Root
            git add data/projects.json 2>&1 | Out-Null
            git commit -m "chore: link $ProjectKey to KR $($merged -join ', ')" 2>&1 | Out-Null
        } catch { Write-Host "okr-link commit skipped: $($_.Exception.Message)" } finally { Pop-Location }
    }
}
```

```powershell
function Get-OkrData {
    $okrsProject = Get-Projects | Where-Object { $_.key -ieq 'OKRs' } | Select-Object -First 1
    if (-not $okrsProject) {
        return [pscustomobject]@{ Available = $false; Stale = $false; Reason = "No OKRs project in registry"; Data = $null; JsonPath = $null }
    }
    $jsonPath = Join-Path $okrsProject.canonicalPath "okr-tracker\okrs.json"
    $yamlPath = Join-Path $okrsProject.canonicalPath "okr-tracker\okrs.yaml"
    if (-not (Test-Path -LiteralPath $jsonPath)) {
        return [pscustomobject]@{ Available = $false; Stale = $false; Reason = "okrs.json not found (run okr.py)"; Data = $null; JsonPath = $jsonPath }
    }
    $stale = $false
    if (Test-Path -LiteralPath $yamlPath) {
        $stale = (Get-Item $yamlPath).LastWriteTimeUtc -gt (Get-Item $jsonPath).LastWriteTimeUtc
    }
    try {
        $data = Get-Content -Raw -Path $jsonPath | ConvertFrom-Json
    } catch {
        return [pscustomobject]@{ Available = $false; Stale = $false; Reason = "okrs.json parse error"; Data = $null; JsonPath = $jsonPath }
    }
    return [pscustomobject]@{ Available = $true; Stale = $stale; Reason = ""; Data = $data; JsonPath = $jsonPath }
}
```

```powershell
function Get-OkrStatusEmoji($Status) {
    switch ("$Status".ToLowerInvariant()) {
        "green" { "🟢" } "yellow" { "🟡" } "red" { "🔴" } "met" { "✅" } default { "⚪" }
    }
}

function Get-OkrStatusOrder($Status) {
    switch ("$Status".ToLowerInvariant()) { "red" { 0 } "yellow" { 1 } "green" { 2 } "met" { 3 } default { 4 } }
}

function New-Cockpit {
    New-Item -ItemType Directory -Force -Path $DashboardDir | Out-Null
    $okr = Get-OkrData
    $projects = Get-Projects

    return $htmlPath
}
```
