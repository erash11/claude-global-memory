```powershell
function Resolve-ProjectPath($Project) {
    $recorded = $Project.canonicalPath
    if ($recorded -and (Test-Path -LiteralPath $recorded)) {
        return [pscustomobject]@{ Path = $recorded; Healed = $false; Resolved = $true; Candidates = @() }
    }
    if ($Project.kind -eq 'vault-note') {
        return [pscustomobject]@{ Path = $recorded; Healed = $false; Resolved = $false; Candidates = @() }
    }
    $targetKey = ConvertTo-ProjectKey $Project.key
    $config = Get-DiscoveryConfig
    $found = New-Object System.Collections.Generic.List[string]
    foreach ($root in $config.Roots) {
        $maxDepth = if ($root.maxDepth) { [int]$root.maxDepth } else { 2 }
        foreach ($d in (Get-ChildDirectoriesBounded $root.path $maxDepth $config.ExcludeDirectoryNames)) {
            if ((ConvertTo-ProjectKey (Split-Path -Leaf $d.Path)) -ieq $targetKey) {
                if (-not ($found -contains $d.Path)) { $found.Add($d.Path) }
            }
        }
    }
    if ($found.Count -eq 1) {
        return [pscustomobject]@{ Path = $found[0]; Healed = $true; Resolved = $true; Candidates = @($found.ToArray()) }
    }
    return [pscustomobject]@{ Path = $recorded; Healed = $false; Resolved = $false; Candidates = @($found.ToArray()) }
}
```

```powershell
function Find-Project($ProjectKey) {
    $projects = Get-Projects
    $project = $projects | Where-Object { $_.key -ieq $ProjectKey } | Select-Object -First 1
    if (-not $project) { return $null }
    $resolved = Resolve-ProjectPath $project
    if ($resolved.Healed) {
        $from = $project.canonicalPath
        $project | Add-Member -NotePropertyName pathHealed -NotePropertyValue $true -Force
        $project | Add-Member -NotePropertyName pathHealedFrom -NotePropertyValue $from -Force
        $project.canonicalPath = $resolved.Path
        Write-Host "path healed: $from -> $($resolved.Path) (run '.\scripts\keystone.ps1 heal $($project.key)' to persist)"
    }
    return $project
}
```

```powershell
function Invoke-Heal($ProjectKey) {
    if (-not (Test-Path $ProjectDataPath)) { Write-Host "No projects.json."; return }
    $json = Get-Content -Raw -Path $ProjectDataPath | ConvertFrom-Json
    $projects = @($json.projects)
    $changed = $false
    foreach ($project in $projects) {
        if ($ProjectKey -and ($project.key -ine $ProjectKey)) { continue }
        $resolved = Resolve-ProjectPath $project
        if ($resolved.Healed) {
            Write-Host "heal: $($project.key): $($project.canonicalPath) -> $($resolved.Path)"
            $project.canonicalPath = $resolved.Path
            $changed = $true
        }
    }
    if (-not $changed) { Write-Host "No drifted paths to heal."; return }
    if ($DryRun) { Write-Host "Dry run: would persist healed paths to data/projects.json."; return }
    [pscustomobject]@{ projects = @($projects) } | ConvertTo-Json -Depth 8 | Set-Content -Path $ProjectDataPath -Encoding UTF8
    Write-Host "Persisted healed paths to data/projects.json"
    if (Test-Path (Join-Path $Root '.git')) {
        try {
            Push-Location $Root
            git add data/projects.json 2>&1 | Out-Null
            git commit -m "fix: heal drifted project path(s) in registry" 2>&1 | Out-Null
            Write-Host "Committed healed registry."
        } catch { Write-Host "Heal commit skipped: $($_.Exception.Message)" }
        finally { Pop-Location }
    }
}
```
